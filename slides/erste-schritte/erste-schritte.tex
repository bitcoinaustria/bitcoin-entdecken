\section{Was ist Bitcoin?}

\begin{frame}{Was ist Bitcoin?}
	\begin{columns}[T]
		\begin{column}{0.25\textwidth}
			\centering
			\roundedimage{pix/was-ist-bitcoin.jpg}{2.5cm}{1}
			\vspace{0.3cm}

			{\small Peer-to-Peer Transfer}
		\end{column}

		\begin{column}{0.7\textwidth}
			\begin{itemize}
				\item \textbf{Digitales Geld} - Elektronisch und funktioniert ohne Banken
				      \begin{itemize}
					      \item Funktioniert komplett elektronisch über Computer
					      \item Keine physischen Münzen oder Scheine
				      \end{itemize}
				\item \textbf{Wertetransfer} - Direkte Übertragung zwischen Nutzer:innen
				      \begin{itemize}
					      \item Person A hat weniger Bitcoin nach dem Transfer
					      \item Person B hat mehr Bitcoin nach dem Transfer
				      \end{itemize}
				\item \textbf{Weltweites Netzwerk} - Keine zentrale Kontrolle
				      \begin{itemize}
					      \item Funktioniert ohne Regierung oder Bank
					      \item \textbf{Keine Zensur} - niemand kann Transaktionen blockieren
				      \end{itemize}
				\item \textbf{Pseudonyme Konten} - Keine persönlichen Daten erforderlich
				      \begin{itemize}
					      \item Keine E-Mail-Adresse oder Passwort nötig
					      \item Nur kryptographische Schlüssel (Adressen)
				      \end{itemize}
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\section{Open Source}

\begin{frame}{Closed Source vs. Open Source Software}
	\textbf{Closed Source} bedeutet:
	\begin{itemize}
		\item Bei den meisten Programmen ist der \textbf{Code nicht einsehbar}
		\item Man \textbf{vertraut} der/dem Entwickler:in blindlings
	\end{itemize}

	\vspace{0.5em}
	\textbf{Open Source} bedeutet:
	\begin{itemize}
		\item Code ist \textbf{verfügbar} und \textbf{überprüfbar}
		\item Programm kann von jedem \textbf{kompiliert} werden
	\end{itemize}

	\vspace{0.5em}
	\textbf{Bitcoin} funktioniert nur als \textbf{Open Source} Technologie

	\vspace{1em}
	\begin{alertblock}{\textcolor{white}{Grundsatz}}
		\vspace{3pt}
		\textbf{Vertrauen} ist gut, \textbf{Kontrolle} ist besser
		\vspace{3pt}
	\end{alertblock}
\end{frame}

\section{Nutzer:in}

\begin{frame}{Nutzer:in}
	\begin{columns}[T]
		\begin{column}{0.25\textwidth}
			\centering
			\roundedimage{pix/nutzer.jpg}{2.5cm}{1}
			\vspace{0.3cm}

			{\small Mobile Wallet}
		\end{column}

		\begin{column}{0.7\textwidth}
			\begin{itemize}
				\item Verwendet \textbf{Computer Code} und erstellt einen zufälligen \textbf{Privaten Schlüssel}
				\item Offline-Prozess: Am Computer selbst \textbf{nicht online}
				\item Berechnet daraus den zugehörigen \textbf{Öffentlichen Schlüssel}
				\item Mit dem Privaten Schlüssel kann bewiesen werden, dass der öffentliche Schlüssel zu einem gehört
				\item Teilt anderen den \textbf{öffentlichen Schlüssel} (Adresse) mit
				      \begin{itemize}
					      \item[\tiny\textcolor{gray}{z.B.}] \scriptsize\texttt{bc1qxy2kgdygjrsqtzq2n0yrf2493p83kkfjhx0wlh}
				      \end{itemize}
				\item Sendet und empfängt \textbf{Bitcoin} über diese Adressen
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\section{Node}

\begin{frame}{Node}
	\begin{columns}[T]
		\begin{column}{0.25\textwidth}
			\centering
			\roundedimage{pix/node.jpg}{2.5cm}{1}
			\vspace{0.3cm}

			{\small Bitcoin Knoten}
		\end{column}

		\begin{column}{0.7\textwidth}
			\begin{itemize}
				\item \textbf{Validierung} - Überprüft alle Transaktionen und Blöcke nach Bitcoin-Protokoll
				      \begin{itemize}
					      \item Verhindert Doppelausgaben (Double Spending)
					      \item Prüft digitale Signaturen und Guthaben
				      \end{itemize}
				\item \textbf{Nachrichtenaustausch} - Kommuniziert kontinuierlich mit anderen Knoten
				      \begin{itemize}
					      \item Empfängt und verbreitet neue Transaktionen
					      \item Synchronisiert Blockchain mit dem Netzwerk
				      \end{itemize}
				\item \textbf{Konsensbildung} - Einigt sich mit anderen Knoten über gültige Regeln
				      \begin{itemize}
					      \item Setzt Bitcoin-Protokollregeln durch
					      \item Bestimmt welche Blockchain-Version korrekt ist
				      \end{itemize}
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\section{Miner}

\begin{frame}{Miner}
	\begin{columns}[T]
		\begin{column}{0.25\textwidth}
			\centering
			\roundedimage{pix/miner.jpg}{2.5cm}{1}
			\vspace{0.3cm}

			{\small Mining Hardware}
		\end{column}

		\begin{column}{0.7\textwidth}
			\begin{itemize}
				\item \textbf{Schürft neue Bitcoin} - Generiert neue Bitcoin als Blockbelohnung
				      \begin{itemize}
					      \item Aktuell \textbf{3,125 BTC} pro Block (nach Halving \textbf{2024})
					      \item Zusätzlich Transaktionsgebühren aus dem Block
				      \end{itemize}
				\item \textbf{Löst kryptografische Rätsel} - Berechnet Proof-of-Work Hashes
				      \begin{itemize}
					      \item Sucht nach gültigem Hash unter dem Target-Wert
					      \item Erster Miner mit Lösung gewinnt den Block
				      \end{itemize}
				\item \textbf{Reguliert Blockzeiten} - Stabilisiert \textbf{10-Minuten}-Rhythmus
				      \begin{itemize}
					      \item Schwierigkeit passt sich alle \textbf{2016 Blöcke} an
					      \item Hält Blockzeit konstant bei ca. \textbf{10 Minuten}
				      \end{itemize}
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\section{Bitcoin Updates}

\begin{frame}{Bitcoin Updates}
	\begin{columns}[T]
		\begin{column}{0.25\textwidth}
			\centering
			\roundedimage{pix/governance.jpg}{2.5cm}{1}
			\vspace{0.3cm}

			{\small Governance}
		\end{column}

		\begin{column}{0.7\textwidth}
			\textbf{Wer hat das "Sagen" im Bitcoin-Netzwerk?}

			\vspace{0.5em}
			Konsens von \textbf{fast allen Teilnehmer:innen} nötig:

			\begin{itemize}
				\item \textbf{Nutzer:innen} - Haben die ultimative Kontrolle
				      \begin{itemize}
					      \item Bei Unzufriedenheit: Vertrauen verlieren und Bitcoin verkaufen
					      \item Bestimmen Bitcoin-Wert durch Marktakzeptanz
				      \end{itemize}
				\item \textbf{Nodes} - Setzen Protokollregeln durch\footnotemark
				      \begin{itemize}
					      \item Müssen neue Software akzeptieren und installieren
					      \item Ohne Node-Unterstützung ist Update wertlos
				      \end{itemize}
				\item \textbf{Miner} - Folgen wirtschaftlichen Anreizen
				      \begin{itemize}
					      \item Müssen profitabel bleiben oder gehen bankrott
					      \item Können Netzwerk nicht alleine kontrollieren
				      \end{itemize}
			\end{itemize}

		\end{column}
	\end{columns}

	\footnotetext{Bier, Jonathan. \textit{The Blocksize War: The battle over who controls Bitcoin's protocol rules.} 2021.}
\end{frame}

\section{Zusammenfassung}

\begin{frame}{Bitcoin: Einfach erklärt}
	\begin{columns}[T]
		\begin{column}{0.25\textwidth}
			\centering
			\roundedimage{pix/zusammenfassung-edited.png}{2.5cm}{1}
			\vspace{0.3cm}

			{\small Bitcoin System}
		\end{column}

		\begin{column}{0.7\textwidth}
			\begin{itemize}
				\item \textbf{Open Source} - Vertrauen durch Kontrolle statt Glaube
				      \begin{itemize}
					      \item Code ist überprüfbar und transparent
				      \end{itemize}
				\item \textbf{Gewaltenteilung} - Drei wichtige Akteur:innen
				      \begin{itemize}
					      \item Nutzer:innen, Nodes und Miner balancieren sich aus
				      \end{itemize}
				\item \textbf{Nutzer:innen} - Erstellen Adressen, senden/empfangen Bitcoin
				      \begin{itemize}
					      \item Private/öffentliche Schlüssel-Kryptografie
				      \end{itemize}
				\item \textbf{Nodes} - Validierung, Nachrichtenaustausch, Konsensbildung
				      \begin{itemize}
					      \item Durchsetzung der Bitcoin-Protokollregeln
				      \end{itemize}
				\item \textbf{Miner} - Schürfen neue Bitcoin (\textbf{3,125 BTC}/Block)
				      \begin{itemize}
					      \item Stabilisieren \textbf{10-Minuten}-Blockzeit
				      \end{itemize}
				\item \textbf{Governance} - Dezentrale Entscheidungsfindung
				      \begin{itemize}
					      \item Niemand kontrolliert Bitcoin alleine
				      \end{itemize}
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}
